\chapter{Topological Spaces and Continuous Functions}

\section{Topological Spaces}


\begin{definition}
    A \emph{\textbf{topology}} on a set $X$ is a collection $\mathcal{T}$ of subsets of $X$ having the following properties, 
    \begin{enumerate}[label=(\arabic*)]
        \item $\emptyset$ and $X$ are in $\mathcal{T}$ 
      \item  The union of the elements of any subcollection of $\mathcal{T}$ is in $\mathcal{T}$
      \item The intersection of the elements of any finite subcollection of $\mathcal{T}$ is in $\mathcal{T}$
    \end{enumerate}

    A set $X$ for which a topology $\mathcal{T}$ has been specified is called a topological space.
\end{definition}
\begin{note}
    A topological space is an ordered pair $(X, \mathcal{T})$
\end{note}
\begin{remark}
    Note that properties (2) and (3) are closely related to the properties of open sets in analysis. In fact this is how we define open sets more generally.
\end{remark}
\begin{definition}
If $X$ is a topological space with topology $\mathcal{T}$ then a subset $U$ of $X$ is an \textbf{open set} of X if $U$ belongs to the collection $\mathcal{T}$.
\end{definition}
\begin{remark}
    This is a definition for open sets in the context of a topology.
\end{remark}
\begin{note}
    Using this terminology we can say that a toplogical space is a set $X$ with a collection of subsets of $X$ called open sets such that $\emptyset$ and $X$ are both open and such that arbitrary unions and finite intersections of open sets are open.
\end{note} 


\begin{eg}
    If $X$ is a set, the collection of all subsets of $X$ is a topology on $X$ called the \textbf{\emph{discrete topology}}. The collection with just $\emptyset$ and $X$ is called the \textbf{\emph{trivial topology}}
\end{eg}


\begin{eg}
    If $X$ is a set and $\mathcal{T}_f$ is the collection of all subsets $U$ of $X$ such that $X -U$ is either finite or all of $X$, then $\mathcal{T}_f$ is a topology on $X$ called the $\textbf{\emph{finite complement topology}}$. First we have $X, \emptyset$ is in $\mathcal{T}_f$, since $X - X$ and $X - \emptyset$ are finite. Now given an indexed family of elements of $\mathcal{T}_f$ i.e. $\{U_{\alpha}\}$ that satisfy this property we need to show (2) and (3) hold.We see, 
    \[
        X - \bigcup U_{\alpha} = \bigcap (X - U_{\alpha})
    \]
    This is finite as each $X - U_{\alpha}$ is finite by assumption of $U$. Similarly given $U_{1}, \dots, U_n$ we see, 
    \[
        X - \bigcap^{n}  U_t = \bigcup^{n} (X - U_i)
    \]
    again the latter is finite as union of finite sets is finite.
\end{eg}

\begin{definition}
    Given $\mathcal{T}$ and $\mathcal{T}'$ are two topologies on $X$. If $\mathcal{T} \subset \mathcal{T}'$. Then $\mathcal{T}'$ is \textbf{\emph{finer}}  than $\mathcal{T}$. If $\mathcal{T}'$  properly contains $\mathcal{T}$ then it's \textbf{\emph{strictly finer}}. We also use the terminology \textbf{\emph{coarser}} and $\textbf{\emph{ striclty coarser}}$ the other way around. $\mathcal{T}$ is \textbf{\emph{comparable}} with $\mathcal{T}'$ if either $\mathcal{T} \subset \mathcal{T}'$ or $\mathcal{T}' \subset \mathcal{T}$
\end{definition}
\begin{note}
    This means that the "finest" topology is the discrete topology and the "coarsest" is the trivial.
\end{note} 

\section{Basis for a Topology}
\begin{definition}
    If $X$ is a set, a \textbf{\emph{basis}} for a topology on $X$ is a collection $\mathcal{B}$ of subsets of $X$ (called the basis elements) such that, 
    \begin{enumerate}
        \item For each $x \in X$, there is at least one basis element $B$ containing $x$.
        \item If $x$ belongs to $B_{1} \cap B_{2}$, then there exists a basis element $B_{3}$ containing $x$ such that $B_{3} \subset B_{1} \cap B_{2}$
    \end{enumerate}
\end{definition}
\begin{remark}
    If $\mathcal{B}$ satisfies the above two condition, then the we define the \textbf{\emph{topology \mathcal{T} generated by \mathcal{B}}} as follows: A subset $U$ of $X$ is said to be open in $X$ (in other words in $\mathcal{T}$) if for each $x \in U$, there is $B \in \mathcal{B}$ such that $x \in B$ and $B \subset U$. Each basis element is itself an element in $\mathcal{T}$.
    \vspace{1em}
    
    Note this is basically saying that the topology generated by \mathcal{B} is, 
    \[
        \mathcal{T} = \left \{\bigcup \mathcal{C} \mid \mathcal{C} \subseteq \mathcal{B}\right \}
    \] 
    in other words we take all possible unions (including infinite unions) of the basis elements.
\end{remark}

\begin{eg}
    If $\mathcal{B}$ is the collection of all circular regions in the plane (interiors of circles). Then $\mathcal{B}$ satisfies both conditions of a basis. In the topology generated, a subset $U$ of the plane is open if every $x$ in $U$ lies in some circular region contained in $U$ (note this implies that the border points cannot be included).
\end{eg}
\begin{eg}
    If $X$ is any set, the collection of all one-point subsets of $X$ is a basis for the discrete topology on $X$.
\end{eg}

\vspace{1em}

\qquad We need to check that the collection $\mathcal{T}$ generated by $\mathcal{B}$ is a topology. First if $U = \emptyset$, then clearly it satisfies our condition and hence is in $\mathcal{T}$ if $U = X$ then clearly for each $x \in U $ there is a basis element $B$ (by definition of a basis) containing $X$ and trivially $B \subset X$ and hence $X \in \mathcal{T}$.

\vspace{1em}

Now take two elements $U_{1}, U_{2}$ in $\mathcal{T}$. We need to show that $U_{1} \cap U_{2}\in\mathcal{T}$ or that for any $x \in U_{1} \cap U_{2}$ there is some $B$ such that $x \in B$ and $B \subset U_{1} \cap U_{2}$. Clearly if $x \in U_{1} \cap U_{2}$, $x \in U_{1}$ and $x \in U_{2}$ so we have $B_{1}, B_{2}$ such that $x \in B_{1}, B_{2}$ and $B_{1} \subset U_{1}$ and $B_{2} \subset U_{2}$ clearly $B_{1} \cap B_{2} \subset U_{1} \cap U_{2}$ and (2) states that if $x \in B_{1} \cap B_{2}$ we have $B_{3}$ s.t $x \in B_{3} \subset B_{1} \cap B_{2} \subset U_{1} \cap U_{2}$. Hence, $U_{1} \cap U_{2}$ satisfies our condition for being in $\mathcal{T}$ and is open. Now, using induction we can show a finite intersection is also open.


\vspace{1em}

Lastly, for arbitrary unions of an indexed family consider $\{U_{a}\}_{a \in J}$. Take $x \in \bigcup U$, this means for some $\alpha$  we have $x \in U_{\alpha}$ so exists a $B$ such that $x \in B$ and $B \subset U_{\alpha} \subset \bigcup U$. Hence, exists $B$ s.t $x \in B$ and $B \subset \bigcup U$ implies $\bigcup U \in \mathcal{T}$.

\begin{lemma}
    If $X$ is a set; and $\mathcal{B}$ a basis for a topology $\mathcal{T}$ on $X$. Then $\mathcal{T}$ equals the collections of all unions of elements of $\mathcal{B}$.
\end{lemma}
\begin{proof}
    A collection of elements of $\mathcal{B}$ must be in $\mathcal{T}$ and union of elements in $\mathcal{T}$ are open, hence any union of basis elements must be in $\mathcal{T}$. Now for any $U \in \mathcal{T}$ for each $x \in U$ we can find $B_i$ s.t $x \in B_x, B_x \subset U$, hence we can write $U = \bigcup_{x  \in U} B_x$ making it a union of basis elements.
\end{proof}

\begin{remark}
    Basically every open set $U$ in $X$ can be expressed as a union of it's basis elements, although not necessarily unique.
\end{remark}
\begin{note}
    This non-uniqueness in the representation of $U$ w.r.t a basis is distinctive from the analogous basis vectors in linear algebra where the representation of a vector w.r.t a basis is unique (i.e the linear combination)
\end{note}


\vspace{1em}
\qquad We can go from basis to a topology, we can also construct a basis given the topology itself.
\begin{lemma}
    If $X$ is a topological space. Suppose $\mathcal{C}$ is a collection of open sets of $X$ such that for each open set $U$ of $X$ and each $x \in U$, there is an element $C \in \mathcal{C}$ such that $x \in C \subset U$. Then $\mathcal{C}$ is a basis for the topology of $X$.
\end{lemma}
\begin{proof}
    We must show that $\mathcal{C}$ is a basis. We have two conditions to check. First clearly for any $x \in X$ we know that $X$ is an open set i.e. $X \in \mathcal{T}$, from the lemma we know there is a $C \in \mathcal{C}$ such that $x \in C \subset X$ thence we satisfy the first condition  of the basis.

    \vspace{1em}
    
    Now let $x \in C_{1} \cap C_{2}$ now since both $C_{1}, C_{2}$ are then $C_{1} \cap C_{2}$ is also open and there is some $C_{3} \in \mathcal{C}$ such that $x \in C_{3} \subset C_{1} \cap C_{2}$.

    \vspace{1em}
    
    Now lastly, we need to show that the basis $\mathcal{C}$ generates the topology $\mathcal{T}$. Let $\mathcal{T'}$ be the topology generated by $\mathcal{C}$. Now let $U \in \mathcal{T'}$. By how we generate this means for any $x \in U$ there is some $C_x \in \mathcal{C}$ such that $x \in C_x \subset U$, but note that we have $C_x \in \mathcal{T}$ which means $U = \bigcup C_x \in \mathcal{T}$. Now on the other side if $U \subset \mathcal{T}$ then for each $x$ we have $C_x \in \mathcal{C}$ such that $x \in C_x \subset U$ but trivially $C_x \in \mathcal{T'}$ and using the same logic we have $U \subset \mathcal{T'}$. Hence we get $\mathcal{T} = \mathcal{T'}$.
\end{proof}

\begin{lemma}
    Let $\mathcal{B}$ and $\mathcal{B'}$ be basis for the topologies $\mathcal{T}$ and $\mathcal{T'}$, on $X$. Then these are equivalent,

    \begin{enumerate}
        \item $\mathcal{T'}$ is finer than $\mathcal{T}$
        \item For each $x \in X$ and each basis element $B \in \mathcal{B}$ containing $x$, there is a basis element $B' \in \mathcal{B'}$ such that  $x \in B' \subset B$.
    \end{enumerate}
\end{lemma}

\begin{remark}
    If a topological space is like a truckload of gravel. The pebbles can be thought of the basis elements; after the pebbles are smashed into smaller pieces, these pieces are the basis elements of the new topology which is obviously finer than the first.
\end{remark}

\section{The Order Topology}
If $X$ is a simply ordered set, then the standard topology generated using the order relation is called the order topology.

\vspace{1em}

\begin{definition}
    Let $X$ be a set with a simple order relation with more than one element. And let $\mathcal{B}$ be the following collection of sets,
    \begin{enumerate}
        \item All open intervals $(a,b)$ in $X$
        \item All intervals of the form $[a_{0}, b)$ where $a_{0}$ is the smallest element of $X$.
        \item All intervals of the form $(a, b_{0}]$ where $b_{0}$ is the largest element of $X$.
    \end{enumerate}

    Then this collection is a basis for $X$ and is called the \textbf{order topology}.
\end{definition}
\begin{remark}
    It is easy to see how this is a basis. For any $x \in X$ if $X$ is the largest or smallest element then (2) and (3) ensures their existence. Else, (1) ensures it as we can choose $(a_{0},b_{0})$ that will contain the element.

    \vspace{1em}
    
    Similarly if $x$ is in the intersection of any two sets, the intersection itself is of the form (1),(2) or (3) and hence is in the basis.
\end{remark}

\begin{eg}
    Consider the order topology of the positive integers. In this case the topology generated is discrete (every singleton set is open) which means any subset of $Z$ is also open.
\end{eg}
\begin{definition}
If $X$ is an ordered set, and $a$ is an element of $X$, there are four subsets of $X$ called the rays determined by $a$.
\begin{align*}
    (a, +\infty) &= \{x \mid x > a\}\\
    (-\infty, a) &= \{x \mid x < a\}\\
    [a, +\infty) &= \{x \mid x \ge a\}\\
    (-\infty, a] &= \{x \mid x \le a\}\\
\end{align*}
\end{definition}

\section{The Product Topology}
If $X,Y$ are topolgical spaces, then if we define a topology on $X \times Y$. 
\begin{definition}
    Let $X$ and $Y$ be topological spaces. The \emph{product topology} on $X \times Y$ is the topology having as basis the collection $\mathcal{B}$ of all sets of the form $U \times V$ where $U$ and $V$ are open subsets of $X$ and $Y$ respectively.
\end{definition}

Firstly $\mathcal{B}$ contains $X \times Y$ and hence all possible elements in $X \times Y$ is contained in some set in $\mathcal{B}$. For the second condition we have the intersection of any two basis element also forms a basis element as, 
\[
    (U_{1} \times V_{1}) \cap (U_{2} \times V_{2}) = (U_{1} \cap U_{2}) \times (V_{1} \cap V_{2})
\]
and since $U_{1} \cap U_{2}$ and $V_{1} \cap V_{2}$ also are open.
\begin{note}
    Note that $\mathcal{B}$ is not itself a topology as the union of two elements is not in the set. One can  imagine two rectangles that overlap a bit, clearly the intersection forms a rectangle however the union doesn't.
\end{note}

\begin{theorem}
    If $\mathcal{B}$ is a basis for $X$ and $\mathcal{C}$ is a basis for $Y$ then, 
    \[
        \mathcal{D} = \{B \times C \mid B \in \mathcal{B}, C \in \mathcal{C}\}
    \]
    forms a basis for the topology of $X \times Y$.
\end{theorem}


\begin{definition}
    Let $\pi_{1}: X \times Y \to X$ be defined as, 
    \[
        \pi_{1}(x, y) = x
    \]
    let $\pi_2 : X \times Y \to Y$ be defined as, 
    \[
        \pi_2(x,y) = y
    \]

    then the maps $\pi_{1}$ and $\pi_{2}$ are called the \textbf{\emph{projections}} of $X \times Y$ onto the first and second factors.
\end{definition}

\vspace{1em}

If $U$ is an open subset of $X$, then then set $\pi_1^{-1}(U)$ is precisely the set $U \times Y$ which is open in $X \times Y$.


\begin{theorem}
    The collection, 
    \[
        \mathcal{S} = \{\pi_{1}^{-1}(U) \mid U \text{ open in }X\} \cup \{\pi_{2}^{-1}(V) \mid V \text{ open in }Y\}
    \]

    is a subbasis for the product topology on $X \times Y$
\end{theorem}
\section{The Subspace Topology}
\begin{definition}
    Let $X$ be a topological space with topology $\mathcal{T}$. If $Y$ is a subset of $X$, the collection, 
    \[
        \mathcal{T_Y} = \{Y \cap U \mid U \in \mathcal{T}\}
    \]
    is a topology on $Y$ called the subspace topology. With this $Y$ is called a subspace of $X$.
\end{definition}

\vspace{1em}

To show $\mathcal{T}_Y$ is a topology first we can choose $U = \emptyset$ which gives us $Y \cap U = \emptyset$ and choosing $U = X$ we get $U \cap Y = Y$ and hence we have $\emptyset$ and $Y$ is in $\mathcal{T}_Y$.

\vspace{1em}

Now for any two sets in $\mathcal{T}_Y$ say $Y \cap U_{1}$ and $Y \cap U_{2}$ we the intersection as $Y \cap U_{1} \cap U_{2}$ or $Y \cap (U_{1} \cap U_{2})$ and clearly $U_{1} \cap U_{2} \in \mathcal{T}$ as $T$ is a topology and hence $Y \cap (U_{1} \cap U_{2})$ is in $\mathcal{T}_Y$. Taking arbitrary unions we get $Y \cap (U_{1} \dots)$ and as $T$ is a topology the arbitrary unions of $U_i$ is open and hence $Y \cap (U_{1} \dots )$ is in $\mathcal{T}_Y$.


\begin{lemma}
   If $\mathcal{B}$ is a basis for the topology of $X$ then the collection, 
   \[
    \mathcal{B}_Y = \{B \cap Y \mid B \in \mathcal{B}\}
   \] 
   is a basis for the subspace topology
\end{lemma}
\begin{proof}
   First we show it's a basis. For any $y \in Y$  we have $y \in X$ and hence some $B$ such that $y \in B$ which implies $y \in B \cap Y$ an element of $\mathcal{B}_Y$. Similarly if we have $y \in (B_{1} \cap Y) \cap (B_{2} \cap Y) $ then we have $y \in Y \cap (B_{1} \cap B_{2})$. This means $y \in B_{1} \cap B_{2}$ and as $B$ is a basis means exists $B_{3}$ such that $y \in B_{3} \subset B_{1} \cap B_{2}$. Hence we have $y \in Y \cap B_{3} \subset Y \cap B_{1} \cap B_{2}$.

   \vspace{1em}
   
    Consider an open set $U$ of $X$ and consider an open set of $Y$ under the subspace topology i.e is of the form $U \cap Y$. Note that we can choose an element $B$ of $\mathcal{B}$ such that $y \in B \subset U$ (as $\mathcal{B}$ is a basis). So we also have $ y\in B \cap Y \subset U \cap Y$. Which makes the above $\mathcal{B}_Y$ a basis for the subspace topology on $Y$.
\end{proof}

\begin{lemma}
    If $Y$ is a subspace of $X$ and if $U$ is open in $Y$ and $Y$ is open in $X$ then $U$ is open in $X$.
\end{lemma}
\begin{proof}
    If $U$ is open in $Y$ then exists $V$ an open set in $X$ such that $U = V \cap Y$. Now finite intersections of open sets are open. Hence $U$ is open in $X$.
\end{proof}
